cmake_minimum_required(VERSION 3.22)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(frdm_mcxn947_robot_controller)
if(CONFIG_ROBOT_GELLO_RADIO AND (CONFIG_ROBOT_GELLO_UART OR CONFIG_ROBOT_GELLO_USB))
  message(FATAL_ERROR "Radio GELLO cannot share the wired GELLO backend")
endif()
if(CONFIG_ROBOT_GELLO_RADIO)
  target_sources(app PRIVATE src/gello/gello_codec.c src/gello/gello_radio_codec.c src/gello/gello_radio_link.c)
endif()
if(CONFIG_ROBOT_GELLO_STANDALONE)
  target_sources(app PRIVATE src/standalone/gello_follow.c src/standalone/gello_standalone.c)
endif()
if(CONFIG_ROBOT_GELLO_UART AND CONFIG_ROBOT_GELLO_USB)
  message(FATAL_ERROR "Choose exactly one board GELLO transport: UART or USB")
endif()
if(CONFIG_ROBOT_GELLO_UART OR CONFIG_ROBOT_GELLO_USB)
  target_sources(app PRIVATE src/gello/gello_codec.c src/gello/gello_uart.c)
endif()
if(CONFIG_ROBOT_GELLO_USB OR CONFIG_ROBOT_GELLO_RADIO_USB)
  include(cmake/gello_usb_ehci.cmake)
  target_sources(app PRIVATE src/gello/gello_usb.c src/gello/gello_usb_desc.c src/gello/gello_usb_clock.c)
  target_include_directories(app PRIVATE ${ZEPHYR_BASE}/subsys/usb/host)
endif()
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(GLOB WL_SOURCES CONFIGURE_DEPENDS "${SDK_ROOT}/third_party/wirelink/src/*.c")
target_include_directories(app PRIVATE src "${SDK_ROOT}/generated/protocol" "${SDK_ROOT}/third_party/wirelink/include")
target_sources(app PRIVATE src/main.c src/core/controller.c src/core/fault_ledger.c src/core/fault_manager.c src/core/clock_reference.c src/devices/can_codec.c src/devices/can_feedback.c src/devices/can_bus.c src/devices/can_trace.c "${SDK_ROOT}/generated/protocol/robot_messages.c" ${WL_SOURCES})

if(CONFIG_ROBOT_PEER_LINK)
  target_sources(app PRIVATE src/peer/peer_codec.c src/peer/peer_link.c)
endif()

target_sources(app PRIVATE src/homing/gravity.c src/homing/home_path.c src/homing/home_still.c src/homing/paired_home.c src/homing/homing.c src/homing/local_home.c)
target_sources(app PRIVATE src/devices/stopped_can.c)
target_sources(app PRIVATE src/devices/hand_transaction.c)
target_sources(app PRIVATE src/devices/hand_startup.c)
target_sources(app PRIVATE src/devices/hand_current.c)

if(CONFIG_ROBOT_GELLO_WEB)
  target_sources(app PRIVATE src/gello/gello_web_codec.c src/gello/gello_action.c)
endif()
