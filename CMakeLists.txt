cmake_minimum_required(VERSION 3.22)
project(panthera_spatial_sdk LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_SHARED_LIBS "Build shared libraries" OFF)
include(CTest)

set(WIRELINK_BUILD_ASIO_UDP_ADAPTER ON)
set(WIRELINK_ASIO_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/asio/include")
add_subdirectory(third_party/wirelink EXCLUDE_FROM_ALL)
set(BLASFEO_TARGET GENERIC CACHE STRING "BLASFEO instruction set for SDK binaries")
# Keep acados from selecting its source checkout as the installation prefix.
set(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT FALSE)
add_subdirectory(third_party/acados EXCLUDE_FROM_ALL)

add_library(panthera_eigen INTERFACE)
target_include_directories(panthera_eigen SYSTEM INTERFACE third_party/eigen)
set(PYBIND11_FINDPYTHON ON)
add_subdirectory(third_party/pybind11 EXCLUDE_FROM_ALL)

add_library(panthera_protocol STATIC
    generated/protocol/robot_messages.c
    generated/protocol/robot_messages_bindings.c
    generated/protocol/robot_messages_runtime.c)
target_include_directories(panthera_protocol PUBLIC generated/protocol)
target_link_libraries(panthera_protocol PUBLIC Wirelink::wirelink)

add_library(panthera_motion src/motion_qp.cpp src/geometry.cpp generated/model/panthera_left.c generated/model/panthera_right.c)
target_include_directories(panthera_motion PUBLIC include PRIVATE generated/model)
target_link_libraries(panthera_motion PUBLIC panthera_eigen PRIVATE acados)
add_library(panthera_telemetry src/telemetry.cpp src/session.cpp)
target_include_directories(panthera_telemetry SYSTEM PRIVATE "${WIRELINK_ASIO_INCLUDE_DIR}")
if(WIN32)
    target_compile_definitions(panthera_telemetry PRIVATE _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(panthera_telemetry PRIVATE ws2_32 mswsock)
endif()
target_include_directories(panthera_telemetry PUBLIC include)
target_link_libraries(panthera_telemetry PUBLIC panthera_protocol PRIVATE Wirelink::asio_udp)
add_executable(panthera-read-state examples/read_state.cpp)
target_link_libraries(panthera-read-state PRIVATE panthera_telemetry)

# Build the QP backend even when the integration test is disabled.
add_custom_target(panthera_host ALL DEPENDS panthera_protocol acados)

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME protocol_provenance
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/ci/check_protocol.py")
    add_executable(geometry_test tests/geometry.cpp)
    target_link_libraries(geometry_test PRIVATE panthera_motion)
    add_test(NAME geometry COMMAND geometry_test)
    add_executable(motion_qp_test tests/motion_qp.cpp)
    target_link_libraries(motion_qp_test PRIVATE panthera_motion)
    add_test(NAME motion_qp COMMAND motion_qp_test)
    add_executable(session_client_test tests/session_client.cpp)
    target_include_directories(session_client_test PRIVATE src)
    if(WIN32)
        target_compile_definitions(session_client_test PRIVATE _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN NOMINMAX)
    endif()
    target_link_libraries(session_client_test PRIVATE panthera_telemetry)
    add_test(NAME sim_session
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/sim_session.py" $<TARGET_FILE:session_client_test>)
    set_tests_properties(sim_session PROPERTIES TIMEOUT 65)
    add_executable(telemetry_capture_test tests/telemetry_capture.cpp)
    target_link_libraries(telemetry_capture_test PRIVATE panthera_telemetry)
    add_test(NAME sim_telemetry
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/sim_telemetry.py" $<TARGET_FILE:telemetry_capture_test>)
    set_tests_properties(sim_telemetry PROPERTIES TIMEOUT 15)
    add_executable(telemetry_loopback_test tests/telemetry_loopback.cpp)
    target_include_directories(telemetry_loopback_test SYSTEM PRIVATE "${WIRELINK_ASIO_INCLUDE_DIR}")
    target_link_libraries(telemetry_loopback_test PRIVATE panthera_telemetry)
    if(WIN32)
        target_compile_definitions(telemetry_loopback_test PRIVATE _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(telemetry_loopback_test PRIVATE ws2_32 mswsock)
    endif()
    add_test(NAME telemetry_loopback COMMAND telemetry_loopback_test)
    add_executable(host_dependencies_test tests/host_dependencies.cpp)
    target_link_libraries(host_dependencies_test PRIVATE panthera_protocol panthera_eigen acados)
    add_test(NAME host_dependencies COMMAND host_dependencies_test)
endif()
